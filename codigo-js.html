JavaScript para Minecraft Bedrock

📌 Requisitos previos

Para desarrollar con JavaScript en Minecraft necesitas:

  • Minecraft Bedrock Edition (versión 1.20.30+)
  • Conocimientos básicos de JavaScript
  • Un editor de código (VS Code recomendado)
  • Habilitar "GameTest Framework" en las configuraciones experimentales

Estructura Básica de un Script

// Importar módulos necesarios
import { world, system } from "@minecraft/server";
import { ActionFormData } from "@minecraft/server-ui";

// Evento cuando un jugador se une
world.afterEvents.playerSpawn.subscribe((event) => {
    const player = event.player;
    
    // Mensaje de bienvenida
    player.sendMessage(`§a¡Bienvenido ${player.name} al servidor!`);
    
    // Dar item inicial
    player.runCommand("give @s diamond 1");
});

Crear una UI Interactiva

async function showPlayerMenu(player) {
    // Crear formulario
    const menu = new ActionFormData()
        .title("Menú del Servidor")
        .body("Selecciona una opción:")
        .button("Ver Perfil", "textures/ui/perfil")
        .button("Tienda", "textures/ui/tienda") 
        .button("Minijuegos", "textures/ui/juegos")
        .button("Cerrar");

    // Mostrar menú
    const response = await menu.show(player);

    // Manejar selección
    switch(response.selection) {
        case 0:
            showProfile(player);
            break;
        case 1:
            showShop(player);
            break;
        case 2:
            showGames(player);
            break;
    }
}

Manejo de Eventos

// Escuchar cuando un jugador rompe un bloque
world.afterEvents.blockBreak.subscribe((event) => {
    const player = event.player;
    const block = event.block;
    
    player.sendMessage(`§7Rompiendo: ${block.typeId}`);
    
    // Dar experiencia según el bloque
    if (block.typeId === "minecraft:diamond_ore") {
        player.addExperience(10);
        player.sendMessage("§a+10 XP");
    }
});

// Sistema de cooldown para comandos
const cooldowns = new Map();

world.beforeEvents.itemUse.subscribe((event) => {
    const player = event.source;
    const item = event.itemStack;
    
    if (item?.typeId === "minecraft:stick") {
        const lastUse = cooldowns.get(player.id) || 0;
        const now = Date.now();
        
        if (now - lastUse < 5000) {
            event.cancel = true;
            player.sendMessage("§cEspera 5 segundos para usar esto");
        } else {
            cooldowns.set(player.id, now);
            player.sendMessage("§a¡Has usado el item especial!");
        }
    }
});

Comandos Personalizados

import { world, system } from "@minecraft/server";

// Registrar comando personalizado
world.beforeEvents.chatSend.subscribe((event) => {
    const message = event.message;
    const player = event.sender;
    
    // Comando /fly
    if (message.startsWith("!fly")) {
        event.cancel = true;
        
        // Verificar permisos
        if (hasPermission(player, "fly")) {
            toggleFly(player);
        } else {
            player.sendMessage("§cNo tienes permiso para usar este comando");
        }
    }
});

function toggleFly(player) {
    const ability = player.getComponent("minecraft:abilities");
    
    if (ability?.canFly) {
        player.runCommand("ability @s mayfly false");
        player.sendMessage("§cModo vuelo desactivado");
    } else {
        player.runCommand("ability @s mayfly true");
        player.sendMessage("§aModo vuelo activado");
    }
}

Sistema de Puntos y Economía

// Base de datos simple con propiedades dinámicas
class EconomySystem {
    constructor() {
        this.balances = new Map();
    }
    
    async getBalance(player) {
        const property = player.getDynamicProperty("balance");
        return property || 0;
    }
    
    async addBalance(player, amount) {
        const current = await this.getBalance(player);
        const newBalance = current + amount;
        
        player.setDynamicProperty("balance", newBalance);
        player.sendMessage(`§a+${amount} monedas. Total: ${newBalance}`);
        
        return newBalance;
    }
    
    async removeBalance(player, amount) {
        const current = await this.getBalance(player);
        
        if (current >= amount) {
            const newBalance = current - amount;
            player.setDynamicProperty("balance", newBalance);
            player.sendMessage(`§c-${amount} monedas. Total: ${newBalance}`);
            return true;
        }
        
        player.sendMessage("§cSaldo insuficiente");
        return false;
    }
}

// Uso del sistema
const economy = new EconomySystem();

world.afterEvents.playerSpawn.subscribe(async (event) => {
    const player = event.player;
    const balance = await economy.getBalance(player);
    
    player.sendMessage(`Tu saldo: §a${balance} monedas`);
});

Ejemplo: Sistema de Recompensas Diarias

class DailyRewards {
    constructor() {
        this.rewards = [
            { item: "minecraft:diamond", amount: 1, day: 1 },
            { item: "minecraft:emerald", amount: 5, day: 2 },
            { item: "minecraft:netherite_ingot", amount: 1, day: 3 },
            { item: "minecraft:experience_bottle", amount: 10, day: 4 },
            { item: "minecraft:elytra", amount: 1, day: 5 }
        ];
    }
    
    async claimDaily(player) {
        const lastClaim = player.getDynamicProperty("last_claim");
        const streak = player.getDynamicProperty("streak") || 0;
        const now = Date.now();
        
        // Verificar si ya reclamó hoy (24 horas)
        if (lastClaim && (now - lastClaim) < 86400000) {
            const hoursLeft = Math.ceil((86400000 - (now - lastClaim)) / 3600000);
            player.sendMessage(`§cPróxima recompensa en ${hoursLeft} horas`);
            return;
        }
        
        // Calcular día de la racha
        let currentDay = streak + 1;
        if (currentDay > this.rewards.length) {
            currentDay = 1;
        }
        
        // Dar recompensa
        const reward = this.rewards[currentDay - 1];
        player.runCommand(`give @s ${reward.item} ${reward.amount}`);
        
        // Actualizar datos
        player.setDynamicProperty("last_claim", now);
        player.setDynamicProperty("streak", currentDay);
        
        player.sendMessage(`§a¡Recompensa del día ${currentDay} reclamada!`);
        player.sendMessage(`§7Has recibido: ${reward.amount}x ${reward.item}`);
    }
}

// Uso
const daily = new DailyRewards();

// Comando !daily
world.beforeEvents.chatSend.subscribe((event) => {
    if (event.message === "!daily") {
        event.cancel = true;
        daily.claimDaily(event.sender);
    }
});
