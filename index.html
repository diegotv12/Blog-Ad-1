Ejemplo: UI Básica con JavaScript

// Ejemplo de UI para Minecraft Bedrock
import * as mc from "@minecraft/server-ui";

async function showSimpleUI(player) {
    const form = new mc.ActionForm()
        .title("Menú Principal")
        .body("Bienvenido al servidor")
        .button("Ver Stats", "textures/ui/stats")
        .button("Tienda", "textures/ui/shop")
        .button("Cerrar");

    const response = await form.show(player);
    
    if (response.selection === 0) {
        player.sendMessage("Tus stats se muestran aquí");
    }
}
