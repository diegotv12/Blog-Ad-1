Configuración de Servidor Bedrock

Paso 1: Descargar el servidor

Descarga la última versión de Minecraft Bedrock Dedicated Server desde la página oficial de Minecraft.

# Estructura de archivos del servidor
bedrock_server/
├── bedrock_server.exe
├── bedrock_server
├── server.properties
├── permissions.json
├── whitelist.json
└── worlds/
    └── Bedrock level/

Paso 2: Configurar server.properties

# Configuración básica del servidor
server-port=19132
server-name=Minecraft Dev Server
max-players=20
gamemode=survival
difficulty=normal
allow-cheats=true
online-mode=true
white-list=false

# Configuración de mundo
level-name=Bedrock level
level-seed=
level-type=DEFAULT

# Configuración de red
max-threads=8
compression-threshold=1
compression-level=6

# Configuración de rendimiento
view-distance=10
tick-distance=4
player-idle-timeout=30

Optimización de Rendimiento

# Parámetros de inicio optimizados (para Linux)
./bedrock_server \
  --max-players 50 \
  --port 19132 \
  --server-name "Optimized Server" \
  --view-distance 12 \
  --tick-distance 6

# Script de inicio avanzado (start.sh)
#!/bin/bash
export LD_LIBRARY_PATH=.
export UE4_LOGGER_SUPPRESS=1
./bedrock_server \
  --max-players $(( $(nproc) * 10 )) \
  --port 19132 \
  --view-distance $(( $(nproc) * 2 )) \
  --tick-distance 4

Respaldo Automático

#!/bin/bash
# Script de backup para servidor

BACKUP_DIR="/backups/minecraft"
WORLD_DIR="/server/worlds"
DATE=$(date +%Y%m%d_%H%M%S)

# Crear directorio de backup
mkdir -p $BACKUP_DIR

# Detener servidor (opcional)
# screen -S minecraft -X stuff "say Servidor en mantenimiento por backup...^M"
# sleep 5
# screen -S minecraft -X stuff "stop^M"
# sleep 10

# Comprimir mundo
tar -czf "$BACKUP_DIR/world_$DATE.tar.gz" $WORLD_DIR

# Iniciar servidor
# screen -S minecraft -X stuff "./start.sh^M"

# Mantener solo últimos 7 backups
find $BACKUP_DIR -name "world_*.tar.gz" -mtime +7 -delete

echo "Backup completado: world_$DATE.tar.gz"

Monitoreo del Servidor

#!/bin/bash
# Script de monitoreo

SERVER_PORT=19132
LOG_FILE="/server/logs/monitor.log"

while true; do
    # Verificar si el servidor está respondiendo
    if nc -z localhost $SERVER_PORT; then
        # Obtener estadísticas
        PLAYERS=$(screen -S minecraft -X stuff "list^M" | tail -n 1)
        MEMORY=$(free -m | grep Mem | awk '{print $3/$2 * 100}')
        CPU=$(top -bn1 | grep "Cpu(s)" | awk '{print $2}')
        
        # Registrar
        echo "$(date): Jugadores: $PLAYERS, Memoria: $MEMORY%, CPU: $CPU%" >> $LOG_FILE
        
        # Alertas
        if (( $(echo "$MEMORY > 90" | bc -l) )); then
            echo "ALERTA: Memoria alta ($MEMORY%)" >> $LOG_FILE
        fi
    else
        echo "$(date): ERROR - Servidor no responde" >> $LOG_FILE
        # Reiniciar servidor si es necesario
        # ./start.sh
    fi
    
    sleep 60
done
